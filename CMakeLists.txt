# Build identity is refreshed for every build, including local commits/source edits
# that do not rerun CMake. Each configuration owns its copy-if-different header.
set(overworld_build_info_dir "${CMAKE_CURRENT_BINARY_DIR}/build-info/$<CONFIG>")
set(overworld_voice_flavor voice-off)
if(OVERWORLD_VOICE_WHISPER)
    if(OVERWORLD_VOICE_AVX2 AND CMAKE_GENERATOR_PLATFORM STREQUAL "x64")

    endif()

add_custom_target(overworld_build_info
    COMMAND "${CMAKE_COMMAND}"
        "-DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}"
        "-DOUTPUT_FILE=${overworld_build_info_dir}/BuildInfo.h"
        "-DAPP_VERSION=${PROJECT_VERSION}"
        "-DBUILD_ARCH=${CMAKE_GENERATOR_PLATFORM}"
        "-DBUILD_CONFIG=$<CONFIG>"
        "-DBUILD_FLAVOR=${overworld_voice_flavor}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/WriteBuildInfo.cmake"
    BYPRODUCTS "${overworld_build_info_dir}/BuildInfo.h"
    VERBATIM)
add_dependencies(Overworld overworld_build_info)
target_include_directories(Overworld PRIVATE "${overworld_build_info_dir}")
