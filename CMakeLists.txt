# Build identity is refreshed for every build, including local commits/source edits
# that do not rerun CMake. Each configuration owns its copy-if-different header.
set(gamepad_build_info_dir "${CMAKE_CURRENT_BINARY_DIR}/build-info/$<CONFIG>")
set(gamepad_voice_flavor voice-off)
if(GAMEPAD_VOICE_WHISPER)
    if(GAMEPAD_VOICE_AVX2 AND CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
        set(gamepad_voice_flavor voice-avx2)
    else()
        set(gamepad_voice_flavor voice-portable)
    endif()
endif()
add_custom_target(gamepad_build_info
    COMMAND "${CMAKE_COMMAND}"
        "-DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}"
        "-DOUTPUT_FILE=${gamepad_build_info_dir}/BuildInfo.h"
        "-DAPP_VERSION=${PROJECT_VERSION}"
        "-DBUILD_ARCH=${CMAKE_GENERATOR_PLATFORM}"
        "-DBUILD_CONFIG=$<CONFIG>"
        "-DBUILD_FLAVOR=${gamepad_voice_flavor}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/WriteBuildInfo.cmake"
    BYPRODUCTS "${gamepad_build_info_dir}/BuildInfo.h"
    VERBATIM)
add_dependencies(GamepadDesktop gamepad_build_info)
target_include_directories(GamepadDesktop PRIVATE "${gamepad_build_info_dir}")
