cmake_minimum_required(VERSION 3.24)
project(GamepadWhisperLibrary LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_C_COMPILER_ID STREQUAL "Clang"
   OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang"
   OR NOT CMAKE_C_COMPILER_VERSION VERSION_EQUAL "20.1.8"
   OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "20.1.8"
   OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This dependency-only project requires clang-cl 20.1.8 targeting Microsoft x64/ARM64 ABI.")
endif()
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release"
   OR NOT CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "MultiThreaded")
    message(FATAL_ERROR "Only Release with the static Microsoft CRT (/MT) is supported.")
endif()
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
foreach(option BUILD_SHARED_LIBS WHISPER_BUILD_TESTS WHISPER_BUILD_EXAMPLES WHISPER_BUILD_SERVER
        WHISPER_CURL WHISPER_COREML WHISPER_OPENVINO GGML_BUILD_TESTS GGML_BUILD_EXAMPLES
        GGML_CUDA GGML_MUSA GGML_HIP GGML_VULKAN GGML_KOMPUTE GGML_METAL GGML_SYCL GGML_OPENCL
        GGML_BLAS GGML_ACCELERATE GGML_RPC GGML_OPENMP GGML_BACKEND_DL GGML_CPU_ALL_VARIANTS
        GGML_CPU_KLEIDIAI GGML_CPU_HBM GGML_NATIVE GGML_LTO GGML_CCACHE
        GGML_SSE42 GGML_AVX GGML_AVX2 GGML_AVX_VNNI GGML_FMA GGML_F16C GGML_BMI2
        GGML_AVX512 GGML_AVX512_VBMI GGML_AVX512_VNNI GGML_AVX512_BF16
        GGML_AMX_TILE GGML_AMX_INT8 GGML_AMX_BF16)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "" CACHE STRING "" FORCE)
include(FetchContent)
if(NOT GAMEPAD_WHISPER_ARCHIVE)
    set(GAMEPAD_WHISPER_ARCHIVE "https://codeload.github.com/ggml-org/whisper.cpp/zip/a8d002cfd879315632a579e73f0148d06959de36")
endif()
FetchContent_Declare(whisper
    URL "${GAMEPAD_WHISPER_ARCHIVE}"
    URL_HASH SHA256=4f02191b61135324a80137e459cd239e037ef3852bad4e6a2ea69490f3d89490
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE TIMEOUT 60)
FetchContent_MakeAvailable(whisper)
add_custom_target(gamepad_whisper_package ALL
    COMMAND "${CMAKE_COMMAND}"
        "-DPACKAGE_DIR=${CMAKE_BINARY_DIR}/package"
        "-DWHISPER_SOURCE=${whisper_SOURCE_DIR}"
        "-DWHISPER_LIB=$<TARGET_FILE:whisper>"
        "-DGGML_LIB=$<TARGET_FILE:ggml>"
        "-DGGML_CPU_LIB=$<TARGET_FILE:ggml-cpu>"
        "-DGGML_BASE_LIB=$<TARGET_FILE:ggml-base>"
        "-DARCH=${GAMEPAD_WHISPER_ARCH}"
        "-DLLVM_VERSION=${CMAKE_CXX_COMPILER_VERSION}"
        "-DMSVC_ABI_VERSION=${CMAKE_CXX_SIMULATE_VERSION}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/WritePackage.cmake"
    DEPENDS whisper ggml ggml-cpu ggml-base VERBATIM)
